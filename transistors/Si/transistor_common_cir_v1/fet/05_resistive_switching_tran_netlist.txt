.libpath "C:\PROGRA~1\QSPICE\EPC"

V_BUS bus 0 {POL*VBUS}
R_LOAD bus d_feed {R_LOAD}
V_ID d_feed d 0
X_DUT d gate s k_dut DUT_FET
V_IK k k_dut 0
L_S s s_return {L_SOURCE}
R_S s_return 0 {R_SOURCE}
V_CMD cmd 0 PULSE(0 1 {T_DELAY} {T_EDGE} {T_EDGE} {TON} {TPER})
R_RET_K driver_return k R={if(KELVIN_RETURN,R_RETURN_ON,R_RETURN_OFF)}
R_RET_0 driver_return 0 R={if(KELVIN_RETURN,R_RETURN_OFF,R_RETURN_ON)}
B_DRV driver driver_return V={POL*(VGS_OFF+(VGS_ON-VGS_OFF)*V(cmd))}
V_IG g_feed gate 0
B_VDS vds 0 V={POL*V(d,k_dut)}
B_VDSP vds_power 0 V={POL*V(d,s)}
B_VSK vs_minus_k 0 V={POL*V(s,k_dut)}
B_VGS vgs 0 V={POL*V(gate,k_dut)}
B_VDRV vdriver 0 V={POL*V(driver,driver_return)}
B_ID id 0 V={POL*I(V_ID)}
B_IG ig 0 V={POL*I(V_IG)}
B_DN vd_ratio 0 V={V(vds)/VBUS}
B_GN vg_ratio 0 V={(V(vgs)-VGS_OFF)/(VGS_ON-VGS_OFF)}
B_PD p_drain 0 V={V(d,s)*I(V_ID)}
B_PG p_gate 0 V={V(gate,s)*I(V_IG)+V(k_dut,s)*I(V_IK)}
B_PT p_total 0 V={V(p_drain)+V(p_gate)}
B´R_G g_feed driver R={if(V(cmd)>0.5,R_GON,R_GOFF)}
.temp 25
.param VBUS=24
.param R_LOAD=48
.param VGS_OFF=0
.param VGS_ON=4.5
.param R_GON=6.8
.param R_GOFF=6.8
.param L_SOURCE=1n
.param R_SOURCE=1m
.param KELVIN_RETURN=0
.param R_RETURN_ON=1u
.param R_RETURN_OFF=1T
.param FSW=100k
.param DUTY=0.5
.param T_DELAY=1u
.param T_EDGE=2n
.param N_CYCLES=4
.param T_MAX=1n
.param TPER={1/FSW}
.param TON={DUTY*TPER-T_EDGE}
.param T_STOP={T_DELAY+N_CYCLES*TPER}
.param T_AVG0={T_DELAY+(N_CYCLES-1)*TPER}
.options reltol=1e-5 abstol=1e-12 vntol=1e-8
.tran 0 {T_STOP} 0 {T_MAX}
.plot V(cmd) V(vds) V(vds_power) V(vs_minus_k) V(vgs) V(vdriver) V(id) V(ig) V(vd_ratio) V(vg_ratio) V(p_drain) V(p_gate) V(p_total)
.meas TRAN TDON_CMD TRIG V(cmd)=0.1 RISE=1 TARG V(vd_ratio)=0.9 FALL=1
.meas TRAN TDON_GATE TRIG V(vg_ratio)=0.1 RISE=1 TARG V(vd_ratio)=0.9 FALL=1
.meas TRAN TR_DRAIN TRIG V(vd_ratio)=0.9 FALL=1 TARG V(vd_ratio)=0.1 FALL=1
.meas TRAN TDOFF_CMD TRIG V(cmd)=0.9 FALL=1 TARG V(vd_ratio)=0.1 RISE=1
.meas TRAN TDOFF_GATE TRIG V(vg_ratio)=0.9 FALL=1 TARG V(vd_ratio)=0.1 RISE=1
.meas TRAN TF_DRAIN TRIG V(vd_ratio)=0.1 RISE=1 TARG V(vd_ratio)=0.9 RISE=1
.meas TRAN P_DRAIN_AVG AVG V(p_drain) FROM={T_AVG0} TO={T_STOP}
.meas TRAN P_GATE_AVG AVG V(p_gate) FROM={T_AVG0} TO={T_STOP}
.meas TRAN P_TOTAL_AVG AVG V(p_total) FROM={T_AVG0} TO={T_STOP}
.meas TRAN VDS_PEAK MAX V(vds)
.meas TRAN ID_PEAK MAX V(id)
.meas TRAN ID_RMS RMS V(id) FROM={T_AVG0} TO={T_STOP}
.lib 00_device.cir
.end
